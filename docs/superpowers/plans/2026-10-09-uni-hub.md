# Uni Hub Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the "Karoheft" uni hub described in `docs/superpowers/specs/2026-10-09-uni-hub-design.md` as a static site served by GitHub Pages from `main` / root.

**Architecture:** Five hand-written HTML pages share one `style.css` and one `main.js`. Theme and splash state live in `localStorage` and are applied by a tiny inline `<head>` script before first paint. No build step. p5.js loads from jsDelivr on the sketch page only.

**Tech Stack:** HTML, CSS custom properties, vanilla JS (ES2020, classic scripts), p5.js 2.3.4 via CDN, Reenie Beanie (OFL) self-hosted.

**Ground rules for every task**
- All visible text is German and is copied verbatim from this plan. It has already gone through the humanizer-de rules: German quotes „…", no dashes as comma replacement, no emojis.
- Asset paths are root-absolute (`/style.css`, `/main.js`, `/fonts/…`). `sketch.js` is the one exception: it stays relative, so a homework folder can be copied as-is.
- No new files beyond those listed. No npm, no frameworks.

// Semester weeks: 01.10.2026 to 31.03.2027 = 26 weeks.
const SEMESTER_WEEKS = 26;

// UTC day math, so daylight saving changes can't shift the count.
function weeksDone(today) {
  const days = (Date.UTC(today.getFullYear(), today.getMonth(), today.getDate()) - Date.UTC(2026, 9, 1)) / 864e5;
  return days < 0 ? 0 : Math.min(SEMESTER_WEEKS, Math.floor(days / 7) + 1);
}

const weeks = document.querySelector('.weeks');
if (weeks) {
  const done = weeksDone(new Date());
  for (let i = 0; i < SEMESTER_WEEKS; i++) {
    const box = document.createElement('li');
    if (i < done) {
      box.className = 'done';
      box.style.setProperty('--i', i);
    }
    weeks.append(box);
  }
  document.getElementById('weeks-label').textContent = `${done} von ${SEMESTER_WEEKS} Wochen abgehakt · M.Sc. Medieninformatik`;
}

// 1D value noise: hashed integer points, smoothstep in between. Returns 0..1.
// ponytail: value noise, not true Perlin; enough for wobbly ink lines.
function noise(x) {
  const hash = (i) => { const s = Math.sin(i * 127.1 + 311.7) * 43758.5453; return s - Math.floor(s); };
  const i = Math.floor(x);
  const f = x - i;
  const t = f * f * (3 - 2 * f);
  return hash(i) + (hash(i + 1) - hash(i)) * t;
}
```

// Ink brush: 7 clicks on the logo (start page only, elsewhere the logo just links home).
let logoClicks = 0;
let brush = null;
let lastPoint = null;
let brushTime = 0;

document.querySelector('.logo').addEventListener('click', (e) => {
  if (location.pathname !== '/') return;
  e.preventDefault();
  if (++logoClicks % 7 || reducedMotion) return;
  if (brush) {
    brush.remove();
    brush = null;
  } else {
    brush = startBrush();
  }
});

// ponytail: canvas keeps its start size, a window resize clips the brush until it's toggled again.
function startBrush() {
  const c = document.createElement('canvas');
  c.id = 'brush';
  c.width = innerWidth;
  c.height = innerHeight;
  document.body.append(c);
  lastPoint = null;
  const g = c.getContext('2d');
  (function fade() {
    if (!c.isConnected) return;
    g.globalCompositeOperation = 'destination-out';
    g.fillStyle = 'rgba(0, 0, 0, .06)';
    g.fillRect(0, 0, c.width, c.height);
    g.globalCompositeOperation = 'source-over';
    requestAnimationFrame(fade);
  })();
  return c;
}

addEventListener('pointermove', (e) => {
  if (!brush) return;
  brushTime += 0.15;
  const p = {
    x: e.clientX + (noise(brushTime) - 0.5) * 14,
    y: e.clientY + (noise(brushTime + 50) - 0.5) * 14,
  };
  if (lastPoint) {
    const g = brush.getContext('2d');
    g.strokeStyle = getComputedStyle(root).getPropertyValue('--ink');
    g.lineWidth = 2.5;
    g.lineCap = 'round';
    g.beginPath();
    g.moveTo(lastPoint.x, lastPoint.y);
    g.lineTo(p.x, p.y);
    g.stroke();
  }
  lastPoint = p;
});
```

```js
// Platzhalter für HA 1. Hier kommt meine eigene Lösung hin.
// Regler gehören in den Sketch: mit .parent(controls) landen sie unter dem Bild.

let scaleSlider;

function setup() {
  createCanvas(720, 300).parent('sketch');
  pixelDensity(1);
  noLoop();

  const controls = select('#sketch-controls');
  createSpan('Skalierung').parent(controls);
  scaleSlider = createSlider(0.002, 0.05, 0.01, 0.001).parent(controls);
  scaleSlider.input(() => redraw());
  createButton('neu würfeln').parent(controls).mousePressed(() => {
    noiseSeed(random(1e6));
    redraw();
  });
}

function draw() {
  const s = scaleSlider.value();
  loadPixels();
  for (let y = 0; y < height; y++) {
    for (let x = 0; x < width; x++) {
      const v = noise(x * s, y * s) * 255;
      const i = 4 * (y * width + x);
      pixels[i] = pixels[i + 1] = pixels[i + 2] = v;
      pixels[i + 3] = 255;
    }
  }
  updatePixels();
  noStroke();
  fill(31, 58, 147);
  textSize(28);
  textAlign(CENTER, CENTER);
  text('Hier kommt meine Lösung hin', width / 2, height / 2);
}
```
